def push_pbft_event(self, phase: str, from_node: int, to_nodes: list, alert_type: str = "") -> None:
    """
    Broadcast a PBFT phase transition to all connected SSE clients.

    Call this from Node.py at each PBFT phase transition:
        self.api.push_pbft_event('pre-prepare', self.node_id, list(peers), alert_type)
        self.api.push_pbft_event('prepare',     self.node_id, list(peers), alert_type)
        self.api.push_pbft_event('commit',      self.node_id, list(peers), alert_type)
        self.api.push_pbft_event('block',       self.node_id, [],          alert_type)
    """
    data = json.dumps({
        "phase":      phase,
        "from_node":  from_node,
        "to_nodes":   to_nodes,
        "alert_type": alert_type,
    })
    with self._sse_lock:
        dead = []
        for q in self._sse_listeners:
            try:
                q.put_nowait(data)
            except queue.Full:
                dead.append(q)
        for q in dead:
            self._sse_listeners.remove(q)
